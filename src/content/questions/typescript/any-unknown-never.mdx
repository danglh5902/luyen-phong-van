answer: "any tắt kiểm tra kiểu; unknown là 'any an toàn' (phải thu hẹp kiểu trước khi dùng); never là kiểu không có giá trị (hàm throw, nhánh không thể xảy ra)."
---

// never: kiểm tra đủ nhánh (exhaustive check)
type Shape = 'circle' | 'square';
function area(s: Shape): number {
  switch (s) {
    case 'circle': return 3.14;
    case 'square': return 1;
    default: { const _x: never = s; return _x; } // thêm Shape mới -> báo lỗi
  }
}
